/help Le client doit envoyer 0
/login [\"user_name\"] Le client doit envoyer 1 \ if log -> serv ret -1
/logout Le client doit envoyer 2 \ if logout -> serv ret -2
/users Le client doit envoyer 3 \ if users -> serv ret -3 + char *users(separator = ;)
/user [\"user_uuid\"] Le client doit envoyer 4 \ if user -> serv ret -4 + char *user 
/send [\"user_uuid\"] [\"message_body\"] Le client doit envoyer 5 \if send -> serv ret -5 
/messages [\"user_uuid\"] Le client doit envoyer 6 \if messages -> serv ret -6 + char *message(separator = ;)
/subscribe [\"team_uuid\"] Le client doit envoyer 7 \if subscribe -> serv ret -7 + char *team else 7 + char *falseteams
/subscribed ?[\"team_uuid\"] Le client doit envoyer 8 \ if subscribed -> serv ret -8 + char *sublist(separator = ;)
/unsubscribe [\"team_uuid\"] Le client doit envoyer 9 \if unsubscribed -> serv ret -9 + char *unsub else 9 + char *falseunsub
/use ?[\"team_uuid\"] ?[\"channel_uuid\"] ?[\"thread_uuid\"] Le client doit envoyer 11
/create Le client doit envoyer 11
/list Le client doit envoyer 12
/info Le client doit envoyer 13
